    # ── Data Pipeline ──
    data/
        __init__.py
        topology_generators.py         # ALL lattice/mesh generators:
                                       #   Category 1 (hex family): uses existing D2C functions
                                       #   Category 2 (random):
                                       #     - generate_blue_noise_network(size, min_dist)
                                       #     - generate_clustered_network(size, n_clusters, std)
                                       #     - generate_gradient_density_network(size, ratio)
                                       #   Category 3 (lattices with basis, triangulated):
                                       #     - generate_kagome_lattice(size)
                                       #     - generate_square_lattice(size)
                                       #     - generate_snub_square_lattice(size)
                                       #     - generate_truncated_hex_lattice(size)
                                       #     - generate_cairo_lattice(size)
                                       #   Category 4 (non-triangulated + soft edges):
                                       #     - generate_honeycomb_lattice(size, k_soft_ratio)
                                       #     - triangulate_polygon_fan(vertices)
                                       #     - triangulate_polygon_center(vertices)
                                       #     - regularize_non_triangulated(mesh, k_soft_ratio)
        graph_utils.py                 # Edge dedup, node feature init, tri->PyG conversion
                                       #   - deduplicate_edges(simplices) -> edge_index, tri_edge_map

                                       #   - triangulation_to_pyg(tri, rigidities, rest_lengths)

                                       #   - generate_single_sample(...)
                                       #   - generate_dataset(config, n_workers)

        configs/

        __init__.py
        model.py                       # ForwardGNN class

        train.py                       # Training loop

                                       #   - Logging to tensorboard/wandb
                                       #   - Early stopping on val loss

                                       #   - Per-topology-class breakdown

                                       #   - Error vs mesh size, error vs topology

    # ── CVAE Inverse ──
    cvae/
        __init__.py
        encoder.py                     # CVAEEncoder(nn.Module)
                                       #   - NNConv backbone (shared arch with GNN)
                                       #   - Set2Set pool + nu_target embedding
                                       #   - -> (mu, logvar), latent_dim=32
        decoder.py                     # CVAEDecoder(nn.Module)
                                       #   - Geometric edge features only (no k, l0)
                                       #   - Node init: geom + z_broadcast + nu_embed
                                       #   - NNConv layers -> edge MLP -> softplus
                                       #   - Outputs: per-edge (k, l0)
                                       #   - edge_mask: freezes soft regularization edges
        model.py                       # FullCVAE: encoder + decoder + loss
                                       #   - L_recon (log-space for k) + beta*L_KL + gamma*L_physics
                                       #   - Beta warmup: 0->1 over 50 epochs
                                       #   - Design variable masking (rig/rl/both)
        train.py                       # CVAE training script
                                       #   - Physics loss via GNN surrogate (fast) then actual solver
        sample.py                      # Inference: sample z ~ N(0,I), decode
        refine.py                      # Optional L-BFGS refinement of CVAE output
                                       #   - Uses Phase 2 differentiable solver
                                       #   - Expected: 5-10 steps vs 40-60 from random
        configs/
            default_cvae_config.yaml
